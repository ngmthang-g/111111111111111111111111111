cmake_minimum_required(VERSION 3.24)
project(TlmSemanticBridge VERSION 2.1.2 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "TLM semantic bridge is Windows-only")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "TLM semantic bridge must be built x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(TlmSemanticBridge SHARED TlmSemanticBridge.cpp)
target_include_directories(TlmSemanticBridge PRIVATE include)
target_compile_definitions(TlmSemanticBridge PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(TlmSemanticBridge PRIVATE user32 kernel32)
if(MSVC)
  target_compile_options(TlmSemanticBridge PRIVATE /utf-8 /W4 /permissive- /O2 /GS /sdl)
  target_link_options(TlmSemanticBridge PRIVATE /OPT:REF /OPT:ICF /DYNAMICBASE /HIGHENTROPYVA /NXCOMPAT)
endif()
set_target_properties(TlmSemanticBridge PROPERTIES OUTPUT_NAME "TlmSemanticBridge")
